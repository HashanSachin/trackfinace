-- Smart Finance Cloud Pro - Complete Multi-Country PostgreSQL Setup & Permissions
CREATE TABLE IF NOT EXISTS users (
    id SERIAL PRIMARY KEY,
    username VARCHAR(50) UNIQUE NOT NULL,
    password_hash TEXT NOT NULL,
    role VARCHAR(20) NOT NULL DEFAULT 'user',
    status VARCHAR(20) NOT NULL DEFAULT 'active',
    preferred_currency VARCHAR(30) NOT NULL DEFAULT 'KWD (د.ك)',
    countries_json TEXT DEFAULT '[{"name":"Kuwait","currency":"KWD","flag":"🇰🇼"},{"name":"Sri Lanka","currency":"LKR","flag":"🇱🇰"}]'
);
ALTER TABLE users ADD COLUMN IF NOT EXISTS countries_json TEXT DEFAULT '[{"name":"Kuwait","currency":"KWD","flag":"🇰🇼"},{"name":"Sri Lanka","currency":"LKR","flag":"🇱🇰"}]';

CREATE TABLE IF NOT EXISTS accounts (
    id SERIAL PRIMARY KEY,
    user_id INT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    name VARCHAR(100) NOT NULL,
    type VARCHAR(50) NOT NULL,
    balance NUMERIC NOT NULL DEFAULT 0.0,
    country VARCHAR(60) NOT NULL DEFAULT 'Kuwait',
    currency VARCHAR(20) NOT NULL DEFAULT 'KWD'
);
ALTER TABLE accounts ADD COLUMN IF NOT EXISTS country VARCHAR(60) NOT NULL DEFAULT 'Kuwait';
ALTER TABLE accounts ADD COLUMN IF NOT EXISTS currency VARCHAR(20) NOT NULL DEFAULT 'KWD';

CREATE TABLE IF NOT EXISTS custom_assets (
    id SERIAL PRIMARY KEY,
    user_id INT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    asset_name VARCHAR(100) NOT NULL,
    category VARCHAR(50) NOT NULL,
    estimated_value NUMERIC NOT NULL DEFAULT 0.0,
    country VARCHAR(60) NOT NULL DEFAULT 'Kuwait',
    currency VARCHAR(20) NOT NULL DEFAULT 'KWD'
);
ALTER TABLE custom_assets ADD COLUMN IF NOT EXISTS country VARCHAR(60) NOT NULL DEFAULT 'Kuwait';
ALTER TABLE custom_assets ADD COLUMN IF NOT EXISTS currency VARCHAR(20) NOT NULL DEFAULT 'KWD';

CREATE TABLE IF NOT EXISTS transactions (
    id SERIAL PRIMARY KEY,
    user_id INT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    account_id INT NOT NULL REFERENCES accounts(id) ON DELETE CASCADE,
    tx_date DATE NOT NULL,
    tx_type VARCHAR(20) NOT NULL,
    category VARCHAR(50) NOT NULL,
    amount NUMERIC NOT NULL,
    description TEXT,
    country VARCHAR(60) NOT NULL DEFAULT 'Kuwait',
    currency VARCHAR(20) NOT NULL DEFAULT 'KWD'
);
ALTER TABLE transactions ADD COLUMN IF NOT EXISTS country VARCHAR(60) NOT NULL DEFAULT 'Kuwait';
ALTER TABLE transactions ADD COLUMN IF NOT EXISTS currency VARCHAR(20) NOT NULL DEFAULT 'KWD';

CREATE TABLE IF NOT EXISTS notifications (
    id SERIAL PRIMARY KEY,
    user_id INT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    title VARCHAR(100) NOT NULL,
    message TEXT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    is_read BOOLEAN DEFAULT FALSE
);

CREATE TABLE IF NOT EXISTS budgets (
    id SERIAL PRIMARY KEY,
    user_id INT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    category VARCHAR(50) NOT NULL,
    monthly_limit NUMERIC NOT NULL,
    year_month VARCHAR(7) NOT NULL,
    country VARCHAR(60) NOT NULL DEFAULT 'Kuwait'
);
ALTER TABLE budgets ADD COLUMN IF NOT EXISTS country VARCHAR(60) NOT NULL DEFAULT 'Kuwait';

CREATE TABLE IF NOT EXISTS savings_goals (
    id SERIAL PRIMARY KEY,
    user_id INT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    goal_name VARCHAR(100) NOT NULL,
    target_amount NUMERIC NOT NULL,
    current_amount NUMERIC NOT NULL DEFAULT 0.0,
    country VARCHAR(60) NOT NULL DEFAULT 'Kuwait'
);
ALTER TABLE savings_goals ADD COLUMN IF NOT EXISTS country VARCHAR(60) NOT NULL DEFAULT 'Kuwait';

-- Disable RLS & Grant API Access to Anon / Publishable Key
ALTER TABLE users DISABLE ROW LEVEL SECURITY;
ALTER TABLE accounts DISABLE ROW LEVEL SECURITY;
ALTER TABLE custom_assets DISABLE ROW LEVEL SECURITY;
ALTER TABLE transactions DISABLE ROW LEVEL SECURITY;
ALTER TABLE notifications DISABLE ROW LEVEL SECURITY;
ALTER TABLE budgets DISABLE ROW LEVEL SECURITY;
ALTER TABLE savings_goals DISABLE ROW LEVEL SECURITY;

GRANT USAGE ON SCHEMA public TO anon, authenticated;
GRANT ALL ON ALL TABLES IN SCHEMA public TO anon, authenticated;
GRANT ALL ON ALL SEQUENCES IN SCHEMA public TO anon, authenticated;
